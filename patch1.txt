<script src="config.js" onerror="window.__configGagal = true; console.warn('config.js gagal dimuat, memakai nilai cadangan di index.html')"></script>
  <script>
    // Tunggu config.js load (sync script, tapi handle error)
    window.configReady = new Promise((resolve) => {
      const check = () => {
        if (window.__configGagal || window.APP_CONFIG) {
          resolve(window.APP_CONFIG || window.APP_DEFAULTS);
        } else {
          setTimeout(check, 10);
        }
      };
      check();
    });
  </script>
